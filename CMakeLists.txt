cmake_minimum_required(VERSION 3.24)
project(OrbitalOdds VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
foreach(dep IN ITEMS raylib box2d json)
  if(dep STREQUAL "raylib")
    set(folder raylib-5.5)
    set(url https://github.com/raysan5/raylib/archive/refs/tags/5.5.zip)
  elseif(dep STREQUAL "box2d")
    set(folder box2d-3.1.0)
    set(url https://github.com/erincatto/box2d/archive/refs/tags/v3.1.0.zip)
  else()
    set(folder json-3.12.0)
    set(url https://github.com/nlohmann/json/archive/refs/tags/v3.12.0.zip)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${folder}/CMakeLists.txt")
    FetchContent_Declare(${dep} SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${folder}")
  else()
    FetchContent_Declare(${dep} URL ${url} DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  endif()
endforeach()
FetchContent_MakeAvailable(raylib box2d json)
add_library(orbital_core src/game.cpp src/feedback.cpp)
target_include_directories(orbital_core PUBLIC src)
target_link_libraries(orbital_core PUBLIC box2d nlohmann_json::nlohmann_json)
add_executable(OrbitalOdds src/main.cpp src/presentation.cpp)
target_link_libraries(OrbitalOdds PRIVATE orbital_core raylib)
if(MINGW)
  target_link_options(OrbitalOdds PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(orbital_core PRIVATE /utf-8 /W4)
  target_compile_options(OrbitalOdds PRIVATE /utf-8 /W4)
else()
  target_compile_options(orbital_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(OrbitalOdds PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(orbital_tests tests/simulation_tests.cpp)
  target_link_libraries(orbital_tests PRIVATE orbital_core)
  if(MINGW)
    target_link_options(orbital_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME simulation COMMAND orbital_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/catalog.json")
endif()
